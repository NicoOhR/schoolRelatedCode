\question[1]{}
\begin{enumerate}[label=\alph*)]
\item Let $\{\mathcal{T}_i\} = \{ \mathcal{T}_i | \mathcal{T}_i \text{ is a
  topology on } X, \mathcal{B} \subset \mathcal{T}_i\}$. Such that
  $\mathcal{T} = \bigcap \mathcal{T}_i$

  \begin{enumerate}
    \item Since each $\mathcal{T}_i$ is a topology, both
      $\varnothing$ and $X$ must be in every $\mathcal{T}_i$,
      therefore, the empty and total set are also in the intersection
      of all those topologies, and are therefore included in $\mathcal{T}$

    \item To show that the union of any sets in $\mathcal{T}$ are in
      $\mathcal{T}$ notice that
      \[
        \forall U \in \mathcal{T} \equiv U \in \bigcap \mathcal{T}_i \implies
        \forall \mathcal{T}_i, U \in \mathcal{T}_i
      \]
      That is, for all open sets in $\mathcal{T}$, they must be in
      every topology $\mathcal{T}_i$ in the intersection. From the
      definition of a topology and what was established above, for any
      collection of open sets in $\mathcal{T}$, $\{U_j\}$ and any
      topology in the intersection $\mathcal{T}_i$
      $$
      \bigcup U_j \in \mathcal{T}_i \implies \bigcup U_j \in \mathcal{T}
      $$
    \item To show that a finite intersection of open sets of
      $\mathcal{T}$ is in $\mathcal{T}$, consider a finite collection
      of sets in $\mathcal{T}$, $\{U_j\}$. As we demonstrated in (2),
      every topology must include every $U_j$. As such
      \[
        \forall i, j \bigcap U_j \in \mathcal{T}_i \implies \bigcap U_j
        \in \mathcal{T}
      \]
  \end{enumerate}
  Thus, $\mathcal{T}$ is a topology.
\item $\mathcal{T} = \mathcal{T}_\mathcal{B}$
  \begin{enumerate}
    \item $\mathcal{T} \subseteq \mathcal{T}_\mathcal{B}$ follows
      immediatly since clearly $\mathcal{B} \subset
      \mathcal{T}_\mathcal{B}$ and therefore $\mathcal{T}_\mathcal{B}
      \in \{\mathcal{T}_i\} \implies \bigcap \mathcal{T}_i \subseteq
      \mathcal{T}_\mathcal{B}$
    \item $\mathcal{T}_\mathcal{B} \subseteq \mathcal{T}$, by
      definition of a toplogy generated by a basis, $U \in
      \mathcal{T}_\mathcal{B}$ if $\forall x \in U, \exists B \in
      \mathcal{B}$ such that $x \in B \subset U$, that is, any open
      set in $\mathcal{T}_\mathcal{B}$ can be expressed a union of $B
      \in \mathcal{B}$. Since every $\mathcal{T}_i$ includes
      $\mathcal{B}$, and an arbitrary union of open sets is open,
      then every $\mathcal{T}_i$ includes $U$ as well, and is
      therefore included in $\mathcal{T}$. Thus, every open set in
      $\mathcal{T}_\mathcal{B}$ is included in $\mathcal{T}$ as well,
      and $\mathcal{T}_\mathcal{B} \subseteq \mathcal{T}$
  \end{enumerate}

\end{enumerate}
\question[2]{}
To show that $\mathcal{B}$ forms a basis, we show that it satisfies
the two conditions necessary:
\begin{enumerate}
\item By definition of the basis $\mathcal{B}$, $\forall x \in X$
  there exists at least one $B = B_d(x, r_n) \in \mathcal{B}$. That
  is, there is at least one ball centered on $x$, therefore there is
  a ball such that $x \in B$
\item Let $B_1, B_2 \in \mathcal{B}$ such that $B_1 \cap B_2 \neq
  \varnothing$. Then $\forall x \in B_1 \cap B_2$, we want to find a
  $B_3 \in \mathcal{B}$ such that $B_3 \subset B_1 \cap B_2$. That
  is, for any $n_1, n_2$ and $x_1, x_2$ such that
  \[
    x \in B(x_1, r_{n_1}) \cap B(x_2, r_{n_2})
  \]
  We want to find an $n$ such that
  $$
  B(x, r_n) \subset B(x_1, r_{n_1}) \cap B(x_2, r_{n_2})
  $$
  Since $x$ is in the intersection of the two balls, we have the two
  following conditions which we know are true:
  \[
    d(x, x_1) \leq r_{n_1} \quad d(x,x_2) \leq r_{n_2}
  \]
  Reasonably, a ball contained in the intersection has maximum radius
  \[
    r := \min(r_{n_1} - d(x,x_1), r_{n_2} -d(x,x_2))
  \]
  Then the object becomes to show that we can alwats find an $n \in
  \mathbb{N}$ such that $r_n \le r$. Since $r > 0$, it suffices to
  show that the sequence $r_n$ converges to $0$. First, we show by
  induction that $0 < r_n < 1$ for all $n$, which also guarantees that
  $\sqrt{1 - r_n}$ is well defined. For $n = 1$, $r_1 = \frac{1}{2}$
  satisfies this. Assume $0 < r_n < 1$, then $0 < 1 - r_n < 1$, so
  $1 + \sqrt{1 - r_n} > 1$, and therefore
  \[
    0 < r_{n+1} = \frac{r_n}{1 + \sqrt{1 - r_n}} < r_n < 1
  \]
  Thus $r_n$ is strictly positive, and so bounded below by zero. Further, we
  prove by induction that the sequence is monotonically decreasing:
  \begin{enumerate}
    \item $r_1 = \frac{1}{2}, r_2 =
      \frac{\frac{1}{2}}{1+\sqrt{1-\frac{1}{2}}} =
      \frac{\frac{1}{2}}{1+\sqrt{\frac{1}{2}}} \approx 0.292, r_1 \ge r_2$
    \item Assume this is true for $r_{n} \ge r_{n+1}$
    \item Then for $r_{n+1} = \frac{r_n}{1+\sqrt{1-r_n}}, r_{n+2} =
      \frac{r_{n+1}}{1+\sqrt{1-r_{n+1}}}$, we verify the inequality
      $r_{n+1} \ge r_{n+2}$. This is simpler with a rationalization
      \[
        \frac{r_{n+1}}{1 + \sqrt{1 - r_{n+1}}} \cdot \frac{1 -
        \sqrt{1 - r_{n+1}}}{1 - \sqrt{1 - r_{n+1}}}
        = \frac{r_{n+1}\left(1 - \sqrt{1 - r_{n+1}}\right)}{1 -
        \left(1 - r_{n+1}\right)}
        = \frac{r_{n+1}\left(1 - \sqrt{1 - r_{n+1}}\right)}{r_{n+1}}
        = 1 - \sqrt{1 - r_{n+1}}
      \]
      Which allows us to verify the inequality as such
      \[
        r_{n} \ge r_{n+1} \implies \sqrt{1 - r_n} \le \sqrt{1 -
        r_{n+1}} \implies 1 - \sqrt{1-r_n} \ge 1 - \sqrt{1-r_{n+1}}
        \implies r_{n+1} \ge r_{n+2}
      \]
  \end{enumerate}
  Since the sequence is bounded below as well as monotonically
  decreasing (also, non increasing), the monotonce convergence
  theorem applies, so $r_n$ converges to some value $L$. We now
  verify that $L = 0$. As $n$ gets very large, it becomes true that
  \begin{align*}
    L &= \frac{L}{1 + \sqrt{1 - L}} \\
    (1 + \sqrt{1 - L})L &= L \\
    L\sqrt{1-L} &= 0 \\
    L^2(1-L) &= 0 \\
  \end{align*}
  Which has two solutions $L = 0$ and $L = 1$, but since $r_n$ is
  bounded above by $r_1 = \frac{1}{2}$, the solution $L = 1$ is
  discarded, so it must be that $L = 0$. As such, for any $\varepsilon >
  0$, there exists an $N \in \mathbb{N}$ such that
  \[
    \forall n \ge N \implies r_n < \varepsilon
  \]
  Take $\varepsilon = r$, then we have that
  \[
    B(x, r_n) \subset B_1 \cap B_2
  \]
  as desired, thus, $\mathcal{B}$ is a basis for a topology on $X$.
\end{enumerate}
To show that $\mathcal{T}_\mathcal{B}$ is equivalent to the metric,
we show that a basis element in one can be expressed as basis
elements of the other. We immediatly have that the metric topology
includes all balls with radius $r_n$ for any $n$ and centered on all
$x$, so $\mathcal{T}_\mathcal{B}$ is a subset of the metric topology.
For the other direction, if any ball in the metric topology
\[
B(x,r)
\]
can be constructed out of sets in $\mathcal{B}$, then we are done. To
do this, take any $B(x, r_n) \in \mathcal{B}$ with $r_n \le r$. Then
$\forall y \in B(x,r), y \notin B(x,r_n)$, take the ball around $y$
with radius $r_m$ with $r_m \le r - d(x,y)$, which, from the previous
part of the question, have shown to always be possible. This ball
centered at $y$ is contained entirely in $B(x,r)$. Then the
union of all such balls is open in the topology generated by
$\mathcal{B}$ is exactly the ball $B(x, r)$. Thus, the metric
topology is a subset of $\mathcal{T}_\mathcal{B}$ and the two
topologies are equal.

\question[3]{}
The properties of the preimage can be demonstrated by a chain of
equivalence statements
\begin{enumerate}
\item
  \begin{align*}
    x \in \bigcup f^{-1}(A_i) &\iff \exists i, x \in f^{-1}(A_i) \\
    &\iff \exists i, f(x) \in A_i \\
    &\iff f(x) \in \bigcup A_i \\
    &\iff x \in f^{-1}(\bigcup A_i)
  \end{align*}
  \begin{align*}
    x \in \bigcap f^{-1}(A_i) &\iff \forall i, x \in f^{-1}(A_i) \\
    &\iff \forall i, f(x) \in A_i \\
    &\iff f(x) \in \bigcap A_i \\
    &\iff x \in f^{-1}(\bigcap A_i)
  \end{align*}
\item
  \begin{align*}
    x \in f^{-1}(Y\setminus A) &\iff f(x) \in Y \setminus A \\
    &\iff f(x) \in Y \text{ and } f(x) \notin A \\
    &\iff x \in X \text{ and } x \notin f^{-1}(A) \\
    &\iff x \in X \setminus f^{-1}(A)
  \end{align*}
\item let $C := g^{-1}(B)$
  \begin{align*}
    x \in (g \circ f)^{-1}(B) &\iff (g \circ f)(x) \in B \\
    &\iff f(x) \in C \\
    &\iff x \in f^{-1}(C) \\
    &\iff x \in f^{-1}(g^{-1}(B))
  \end{align*}
\end{enumerate}
