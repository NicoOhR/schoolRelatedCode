\subsection*{Mutual-friend recommendation (MapReduce)}
The input is the adjacency list $A$, an RDD of pairs $(v, F_v)$,
where $F_v$ is the list of $v$'s friends.

\begin{algorithm}[H]
  \caption{\textsc{FriendRec}$(A, u, k)$: first-order friend recommendations}
  \begin{algorithmic}[1]
    \REQUIRE adjacency list $A = \{(v, F_v)\}$, target user $u$,
    number of recommendations $k$
    \STATE $F_u \leftarrow \textsc{Lookup}(A, u)$ \COMMENT{$u$'s friends}
    \STATE $S \leftarrow F_u \cup \{u\}$ \COMMENT{users to exclude
    from the result}
    \STATE \textbf{Filter:} keep $(v, F_v) \in A$ with $v \in S$
    \STATE \textbf{FlatMap:} $(v, F_v) \mapsto \{\, w : w \in F_v
    \,\}$ \COMMENT{friends of $u$'s friends}
    \STATE \textbf{Map:} $w \mapsto (w, 1)$
    \STATE \textbf{ReduceByKey:} $(w, [1, 1, \dots]) \mapsto (w,
    c_w)$ \COMMENT{$c_w$ = mutual friends of $u$ and $w$}
    \STATE \textbf{Filter:} keep $(w, c_w)$ with $w \notin S$
    \COMMENT{drop $u$ and existing friends}
    \RETURN the $k$ pairs with the largest $c_w$
  \end{algorithmic}
\end{algorithm}

\subsection*{Naive Bayes}
The data set chosen for this implementation was the Diplomacy dataset
from the QANTA project. Diplomacy is a strategy game where players
communicate and coordinate and often attempt to decieve one another.
The dataset collects messages from various games and has the players
themselves label if they are being truthful or not.

Each message is tokenized and stop words are removed, which gives a
word list $d$ and a label $y \in \{\text{lie}, \text{truth}\}$.
Word probabilities are smoothed toward an assumed probability of
$0.5$ with weight $w$ (default $w = 1$):
\[
  \tilde{p}(t \mid y) = \frac{0.5\,w + n_t\, p(t \mid y)}{w + n_t},
\]
where $n_t$ is the total number of times $t$ occurs in the training set.

\begin{algorithm}[H]
  \caption{\textsc{Fit}$(D, w)$}
  \begin{algorithmic}[1]
    \REQUIRE training RDD $D = \{(y, d)\}$, smoothing weight $w$
    \STATE $N_{\text{lie}} \leftarrow |\{(y, d) \in D : y = \text{lie}\}|$
    \STATE $N \leftarrow |D|$, \quad $N_{\text{truth}} \leftarrow N -
    N_{\text{lie}}$
    \STATE \textbf{FlatMap:} $(y, d) \mapsto \{\, ((t, y), 1) : t \in d \,\}$
    \STATE \textbf{ReduceByKey:} $((t, y), [1, 1, \dots]) \mapsto
    ((t, y), c_{t,y})$
    \STATE \textbf{Map:} $((t, y), c_{t,y}) \mapsto ((t, y),\ c_{t,y}
    / N_y)$ \COMMENT{$p(t \mid y)$}
    \STATE $P \leftarrow$ collect the result as a map and broadcast it
  \end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
  \caption{\textsc{Predict}$(d)$ for one message (applied with
  \textbf{Map} over the test RDD)}
  \begin{algorithmic}[1]
    \REQUIRE word list $d$; $P$, $N_{\text{lie}}$,
    $N_{\text{truth}}$, $N$, $w$ from \textsc{Fit}
    \STATE $s_{\text{lie}} \leftarrow \log(N_{\text{lie}} / N)$,
    \quad $s_{\text{truth}} \leftarrow \log(N_{\text{truth}} / N)$
    \FOR{each $t \in d$}
    \STATE $p_{\text{lie}} \leftarrow P[(t, \text{lie})]$, \quad
    $p_{\text{truth}} \leftarrow P[(t, \text{truth})]$ \COMMENT{0 if missing}
    \STATE $n_t \leftarrow p_{\text{lie}} N_{\text{lie}} +
    p_{\text{truth}} N_{\text{truth}}$
    \STATE $s_{\text{lie}} \leftarrow s_{\text{lie}} +
    \log\dfrac{0.5\,w + n_t\, p_{\text{lie}}}{w + n_t}$
    \STATE $s_{\text{truth}} \leftarrow s_{\text{truth}} +
    \log\dfrac{0.5\,w + n_t\, p_{\text{truth}}}{w + n_t}$
    \ENDFOR
    \RETURN truth \textbf{if} $s_{\text{truth}} > s_{\text{lie}}$
    \textbf{else} lie
  \end{algorithmic}
\end{algorithm}
